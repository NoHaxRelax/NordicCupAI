def setup(scenario, seed):
    if scenario=='generated':
        core=SimulationCore(seed=seed)
        return core.env,dict(setup='unmodified SimulationCore defaults',encounters=[])
    e=fixture(seed)
    # Restore real stochastic tree/predator generation omitted by the old fixture.
    e.spawn_tree=types.MethodType(Environment.spawn_tree,e)
    e.spawn_predator=types.MethodType(Environment.spawn_predator,e)
    rng=random.Random(seed+19000)
    for i in range(12):
        tree=e.spawn_tree(290+(i%4)*65+rng.uniform(-12,12),475+(i//4)*90+rng.uniform(-12,12))
        tree.grow(rng.uniform(20,35))
    for i in range(28):
        f=e.spawn_fruit(rng.uniform(260,530),rng.uniform(450,760))
        f.grow(rng.uniform(0,40))
    for i,(x,y) in enumerate(((350,565),(390,600),(430,620),(440,540))):
        a=add_agent(e,x,y,150);a.direction=rng.uniform(-math.pi,math.pi)
        a.max_age=60+rng.uniform(0,60)
    if scenario in ('obstacles','river'):
        for _ in range(10):
            # Real-sized obstacles, outside the orchard and introduced predator.
            x,y=rng.uniform(780,1450),rng.uniform(180,960)
            e.obstacles.append(Obstacle(x,y,rng.uniform(30,70),rng.uniform(40,100)))
        e.edges={edge for o in e.obstacles for edge in o.edges}
    if scenario=='river':e.biome_map[1040:1140,:]=River_biome()
    e._update_spatial_grid()
    meta=dict(setup='arranged finite orchard; original replenishment, aging, births and stochastic predators enabled',
        starting_agents=4,starting_energy=150,starting_trees=12,starting_fruits=28,
        initial_fruit_growth='uniform 0..40',initial_tree_growth='uniform 20..35',
        encounters=[dict(t=20.,x=620.,y=600.+rng.uniform(-65,65),heading=math.pi),
                    dict(t=100.,x=650.,y=620.+rng.uniform(-80,80),heading=math.pi)])
    return e,meta


def run_case(job):
    scenario,seed,mode,horizon,settings,record=job
    start=time.perf_counter();e,scenario_meta=setup(scenario,seed)
    policy=SimplePolicy('nursery',seed) if mode=='nursery' else ColonyPolicy(banish=mode not in ('control','oracle-control'),**settings)
    oracle=mode.startswith('oracle')
    state=step_environment(e,[],.1)
    initial_rng_hash=hashlib.sha256(repr(e.rng.getstate()).encode()).hexdigest()
    recorder=None
    if record:
        from recorder import ReplayRecorder
        recorder=ReplayRecorder(e,title=f'{scenario} seed {seed}: {mode}',policy='colony '+mode,seed=seed,
            scenario=scenario,every=5,notes=('True agent poses supplied. ' if oracle else 'Only cached observation DTOs supplied. ')+
                'Live finite fruit, normal tree turnover, reproduction and new predator generation. No remote calls.',
            policy_sha256=POLICY_HASH)
        recorder.capture(force=True)
    metrics=dict(agent_seconds=0.,worker_seconds=0.,worker_exposure_100=0.,worker_exposure_50=0.,all_exposure_100=0.,
        guide_seconds=0.,guide_move_energy=0.,worker_move_energy=0.,guide_target_seconds=0.,
        predator_active_seconds=0.,colony_intrusion_seconds=0.,harvest_energy=0.,fruit_count=0,
        captures=0,guide_captures=0,worker_captures=0,starvation_deaths=0,births=0,peak_agents=len(e.agents),
        predators_spawned=0,physical_boundary_violations=0,guide_harvest_energy=0.)
    events=[];trace=[];release_records=[];seen_events=0;initial_count=e._next_agent_id
    localization_errors=[];group_errors=[];capture_energy=0.;scheduled=set();seen_pred_count=len(e.predators)
    guide_episodes=[];active_episode=None
    for tick in range(int(horizon*10)-1):
        if not e.agents:break
        for i,enc in enumerate(scenario_meta['encounters']):
            if i not in scheduled and e.time+1e-6>=enc['t']:
                p=e.spawn_predator(enc['x'],enc['y'])
                if p:p.direction=enc['heading'] # Normal 0-energy resting newborn.
                scheduled.add(i)
        inputs=state['observations'];poses={a.agent_id:(a.x,a.y,a.direction) for a in e.agents}
        if mode=='nursery':actions=policy(inputs,e.time);roles={s['agent_id']:'worker' for s in inputs}
        else:
            actions=policy(inputs,e.time,oracle_poses=poses if oracle else None);roles=dict(policy.roles)
        assert len(actions)==len(inputs) and len({aid for aid,a in actions})==len(actions)
        assert all(math.isfinite(x) for aid,a in actions for x in (a.move_distance,a.move_direction,a.turn_angle))
        before={a.agent_id:a for a in e.agents};fruits=set(e.fruits);score_before=e.score
        action_t=e.time;guide=policy.guide if mode!='nursery' else None
        for aid,action in actions:
            a=before[aid];old=a.energy
            e.agent_step(aid,action.move_distance,action.move_direction,action.turn_angle,action.spawn_agent)
            cost=old-a.energy-(100 if action.spawn_agent and old-a.energy>=100 else 0)
            metrics['guide_move_energy' if roles.get(aid)=='guide' else 'worker_move_energy']+=cost
        after_action_energy={aid:a.energy for aid,a in before.items()}
        # Read-only occupation measurement using actual predator sensing after
        # agent actions, immediately before its original update. With multiple
        # predators this is a near-synchronous sample, not a claimed action log.
        for p in e.predators:
            if not p.resting:
                metrics['predator_active_seconds']+=.1
                if guide is not None:
                    obs=p.observe(agents=e._get_local_agents(p),edges=e._get_local_edges(p))
                    targets=[o for o in obs if o['type']=='Agent']
                    target=min(targets,key=lambda o:o['distance'])['id'] if targets else None
                    if target==guide:metrics['guide_target_seconds']+=.1
        e.non_agent_step(.1)
        state=dict(observations=[e.get_agent_state(a.agent_id) for a in e.agents])
        penalty=0.
        for aid,a in before.items():
            role=roles.get(aid,'worker');metrics['agent_seconds']+=.1
            metrics['guide_seconds' if role=='guide' else 'worker_seconds']+=.1
            if aid not in e.agents_dict:
                captured=a.energy>0
                if captured:
                    metrics['captures']+=1;metrics['guide_captures' if role=='guide' else 'worker_captures']+=1
                    penalty+=a.energy/100;capture_energy+=a.energy
                else:metrics['starvation_deaths']+=1
                events.append(dict(t=round(e.time,2),kind='capture' if captured else 'energy_death',aid=aid,role=role,energy=round(a.energy,3)))
            nearest=min((math.hypot(a.x-p.x,a.y-p.y) for p in e.predators),default=math.inf)
            if nearest<100:metrics['all_exposure_100']+=.1
            if role=='worker':
                if nearest<100:metrics['worker_exposure_100']+=.1
                if nearest<50:metrics['worker_exposure_50']+=.1
            if a.agent_id in e.agents_dict and not (30+a.size-1e-6<=a.x<=e.width-30-a.size+1e-6 and 30+a.size-1e-6<=a.y<=e.height-30-a.size+1e-6):
                metrics['physical_boundary_violations']+=1
            if role=='guide' and aid in e.agents_dict:
                passive=.1+(a.age*.01 if a.age>a.max_age else 0)
                metrics['guide_harvest_energy']+=max(0,a.energy-after_action_energy[aid]+passive)
        metrics['harvest_energy']+=(e.score-score_before-.1+penalty)*1000
        metrics['fruit_count']+=sum(f not in e.fruits and f.age<=100 for f in fruits)
        metrics['peak_agents']=max(metrics['peak_agents'],len(e.agents))
        if mode!='nursery':
            for pe in policy.events[seen_events:]:
                events.append(pe)
                if pe['kind']=='guide_selected':
                    a=e.agents_dict.get(pe['aid']);workers=[v for v in e.agents if v.agent_id!=pe['aid']]
                    pred=min(e.predators,key=lambda p:dist((p.x,p.y),(a.x,a.y)),default=None) if a else None
                    if a and workers and pred:
                        center=[sum(v.x for v in workers)/len(workers),sum(v.y for v in workers)/len(workers)]
                        active_episode=dict(start=pe['t'],aid=pe['aid'],predator=e.predators.index(pred),worker_center=center,
                            initial_distance=dist(center,(pred.x,pred.y)),max_distance=dist(center,(pred.x,pred.y)),end=None)
                        guide_episodes.append(active_episode)
                if pe['kind'] in ('guide_finished','guide_lost') and active_episode:
                    active_episode['end']=pe['t'];active_episode['end_kind']=pe['kind'];active_episode=None
                if pe['kind']=='release_attempt':
                    aid=pe['aid'];a=e.agents_dict.get(aid)
                    if a:
                        nearest=min(e.predators,key=lambda p:math.hypot(a.x-p.x,a.y-p.y),default=None)
                        release_records.append(dict(t=pe['t'],aid=aid,predator=e.predators.index(nearest) if nearest else None,
                            home=None,return_t=None,actual_predator=[nearest.x,nearest.y] if nearest else None,
                            guide_energy=pe['energy']))
                        # Actual worker centroid is diagnostic only.
                        workers=[v for v in e.agents if roles.get(v.agent_id)=='worker']
                        if workers:release_records[-1]['home']=[sum(v.x for v in workers)/len(workers),sum(v.y for v in workers)/len(workers)]
            seen_events=len(policy.events)
            if active_episode:
                pr=e.predators[active_episode['predator']]
                active_episode['max_distance']=max(active_episode['max_distance'],dist(active_episode['worker_center'],(pr.x,pr.y)))
            for rel in release_records:
                if rel['return_t'] is None and rel['home'] and rel['predator'] is not None and e.time>rel['t']+.5:
                    p=e.predators[rel['predator']]
                    if dist((p.x,p.y),rel['home'])<250:rel['return_t']=round(e.time,2)
            # Compare relative pair distances, invariant to arbitrary local frames.
            if tick%20==0:
                agents=e.agents
                for i,a in enumerate(agents):
                    for b in agents[i+1:]:
                        if a.agent_id not in policy.map.poses or b.agent_id not in policy.map.poses:continue
                        pa,pb=policy.map.poses[a.agent_id],policy.map.poses[b.agent_id]
                        if pa.group==pb.group:group_errors.append(abs(dist(pa.xy(),pb.xy())-dist((a.x,a.y),(b.x,b.y))))
        workers=[a for a in e.agents if roles.get(a.agent_id)=='worker']
        if workers and any(min(dist((p.x,p.y),(a.x,a.y)) for a in workers)<150 for p in e.predators):
            metrics['colony_intrusion_seconds']+=.1
        if recorder:recorder.capture(actions,decisions=getattr(policy,'decisions',{}),inputs=inputs,action_t=action_t)
        if tick%100==0:
            trace.append(dict(t=round(e.time,2),score=round(e.score,3),agents=len(e.agents),predators=len(e.predators),
                energy=round(sum(a.energy for a in e.agents),2),guide=guide,phase=getattr(policy,'phase',None)))
    metrics['births']=e._next_agent_id-initial_count;metrics['predators_spawned']=len(e.predators)
    result=dict(scenario=scenario,seed=seed,mode=mode,horizon=horizon,settings=settings,scenario_parameters=scenario_meta,
        duration=round(e.time,2),score=round(e.score,5),alive=len(e.agents),
        metrics={k:round(v,4) if isinstance(v,float) else v for k,v in metrics.items()},events=events,releases=release_records,guide_episodes=guide_episodes,trace=trace,
        localization_pair_error_mean=round(sum(group_errors)/len(group_errors),3) if group_errors else None,
        localization_pair_error_p95=round(sorted(group_errors)[int(.95*(len(group_errors)-1))],3) if group_errors else None,
        wall_seconds=round(time.perf_counter()-start,2),source_commit=COMMIT,initial_rng_sha256=initial_rng_hash,
        policy_sha256=POLICY_HASH,harness_sha256=HARNESS_HASH,
        platform=platform.platform(),python=platform.python_version(),
        policy_privilege='true agent poses plus ordinary cached observations' if oracle else 'ordinary cached observations and simulation time only')
    if recorder:
        path=OUT/record;result['replay']=str(path.relative_to(ROOT.parent));recorder.save(path,overwrite=True,reason='extinction' if not e.agents else 'horizon reached')
    print(json.dumps({k:result[k] for k in ('scenario','seed','mode','duration','score','alive','wall_seconds')}|
        {'guides':sum(e['kind']=='guide_selected' for e in events),'releases':len(release_records),'captures':metrics['captures']}),flush=True)
    return result


def main():
    p=argparse.ArgumentParser()
    p.add_argument('--scenarios',nargs='+',default=['forest'],choices=['forest','obstacles','river','generated'])
    p.add_argument('--seeds',nargs='+',type=int,default=[1,2,3])
    p.add_argument('--modes',nargs='+',default=['control','banish'],choices=['control','banish','oracle','oracle-control','nursery'])
    p.add_argument('--horizon',type=float,default=180)
    p.add_argument('--workers',type=int,default=1)
    p.add_argument('--out',default='screen.json')
    p.add_argument('--variant',default='rest',choices=['rest','distance'])
    p.add_argument('--min-fuel',type=float,default=300)
    p.add_argument('--population',type=int,default=4)
    p.add_argument('--release-radius',type=float,default=450)
    p.add_argument('--guide-limit',type=float,default=18)
    p.add_argument('--replay',action='store_true')
    p.add_argument('--no-risk-gate',action='store_true')
    a=p.parse_args()
    settings=dict(variant=a.variant,min_fuel=a.min_fuel,population=a.population,release_radius=a.release_radius,guide_limit=a.guide_limit,risk_gate=not a.no_risk_gate)
    jobs=[(scenario,seed,mode,a.horizon,settings,f'{Path(a.out).stem}-{scenario}-{seed}-{mode}.json.gz' if a.replay else None)
        for scenario in a.scenarios for seed in a.seeds for mode in a.modes]
    if a.workers>1:
        with ProcessPoolExecutor(max_workers=a.workers) as pool:results=list(pool.map(run_case,jobs))
    else:results=[run_case(j) for j in jobs]
    data=dict(source_commit=COMMIT,scope='Local only; no competition API, validation, submission, deployment or push.',
        pairing_note='Matched initial seeds. Engine RNG consumption diverges with behavior. Pairing is not common-random-number coupling.',runs=results)
    OUT.mkdir(exist_ok=True,parents=True);(OUT/a.out).write_text(json.dumps(data,indent=2)+'\n')


if __name__=='__main__':main()
