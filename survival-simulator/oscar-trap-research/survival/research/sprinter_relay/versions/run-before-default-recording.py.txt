"""Native-engine relay runner with read-only event observers.

Fixture placement and diagnostics use privileged geometry. RelayController only
receives ordinary cached agent states. Native methods perform every interaction.
Event observers delegate once to the original method, without changing arguments,
return values, energy, RNG, or physics. Source line assertions pin death/food cause.
"""
from __future__ import annotations
import argparse
import hashlib
import inspect
import json
import math
import sys
import types
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT/'research'), str(ROOT/'debugger'), str(Path(__file__).parent)]
from sprinter_decoys import fixture, add_agent, add_pred, observe, ripe_food
from controller import RelayController, MultiRelayController
from recorder import ReplayRecorder
from src.elements.tree import Tree
from src.elements.environment import Environment

OUT = ROOT/'results/sprinter_relay'
SOURCE = 'acfc31a4003a5f91bf11032a02cd98c178ddbd7e'

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def run(*, seed=0, energy=150, gap=60, heading=0., pred_energy=102., seconds=45,
        policy='mpc', reserve_angle=math.pi, reserve_gap=60, food='none', workers=0,
        worker_near=False, predators=1, pairs=1, record=None, **params):
    parameters = dict(seed=seed, energy=energy, gap=gap, heading=heading,
        pred_energy=pred_energy, seconds=seconds, policy=policy,
        reserve_angle=reserve_angle, reserve_gap=reserve_gap, food=food,
        workers=workers, worker_near=worker_near, predators=predators, pairs=pairs, **params)
    e = fixture(seed)
    baits, groups = [], []
    for index in range(predators):
        cy = 600 + index*350
        add_pred(e, x=800, y=cy, heading=heading, energy=pred_energy)
        # Draw the first bait exactly as the earlier fixtures did, even for none.
        if index == 0 or (pairs > index and policy != 'none'):
            a = add_agent(e, 800+gap, cy, energy)
            if policy == 'none':
                e.kill_agent(a)
                continue
            group = [a.agent_id]
            baits.append(a)
            if policy not in ('single', 'orbit_single'):
                b = add_agent(e, 800+reserve_gap*math.cos(reserve_angle),
                              cy+reserve_gap*math.sin(reserve_angle), energy)
                b.direction = reserve_angle+math.pi
                baits.append(b)
                group.append(b.agent_id)
            groups.append(group)
    worker_ids = []
    for i in range(workers):
        a = add_agent(e, 1050 if worker_near else 1200, 500+70*i, 150)
        a.direction = math.pi
        worker_ids.append(a.agent_id)
    if food == 'finite':
        ripe_food(e, [(740,600),(860,600),(800,660),(800,540),
                     (680,600),(920,600),(800,720),(800,480)] +
                    [(1200,500+70*i) for i in range(workers)])
    if food in ('trees', 'renewing'):
        points = [(1200,500+70*i) for i in range(workers)]
        for index in range(predators if pairs > 1 else 1):
            cy = 600+index*350
            points += [(740,cy),(860,cy),(800,cy+60),(800,cy-60),(680,cy),(920,cy)]
        for x, y in points:
            tree = Tree(x,y)
            tree.grow(20)
            e.trees.append(tree)
        e._update_tree_grid()
        if food == 'renewing':
            e.spawn_tree = types.MethodType(Environment.spawn_tree,e)
    variant = 'orbit' if policy=='orbit_single' else 'mpc' if policy in ('single','none') else policy
    ctrl = (MultiRelayController(groups, variant=variant, **params) if len(groups)>1
            else RelayController([a.agent_id for a in baits], variant=variant, **params))
    hashes = dict(controller_sha256=digest(inspect.getfile(type(ctrl))), runner_sha256=digest(__file__))
    states = observe(e)
    recorder = None
    if record:
        recorder = ReplayRecorder(e,title=record,policy=policy,seed=seed,every=2,
            scenario='arranged relay',policy_sha256=hashes['controller_sha256'],
            notes='Observation-only policy. Arranged forest with physical boundaries. Native energy, food, aging, predator sensing and physics; no hosted calls.')
        recorder.capture(force=True)

    bait_ids = {a.agent_id for a in baits}
    known_ids = set(e.agents_dict)
    births, deaths, switches, trace, ready_events = [], [], [], [], []
    prepared, captured_ids = set(), set()
    pred_active, pred_occupied, pred_gap, pred_max_gap, last_target = {}, {}, {}, {}, {}
    food_count = dict(bait=0,worker=0)
    food_energy = dict(bait=0.,worker=0.)
    food_received = dict(bait=0.,worker=0.)
    spent = dict(bait=0.,worker=0.)
    worker_seconds = population_seconds = birth_cost = captured_energy = 0.
    worker_target_ticks = 0
    peak_population = len(e.agents)
    first_gap = first_capture = None
    min_separation = math.inf
    selected = [None]*predators
    food_energy_cursor = {}

    # These observers inspect the engine's real events, not a parallel simulated
    # interaction pass. They never supply state to the controller.
    native_kill, native_remove = e.kill_agent, e.remove_fruit
    def kill_observer(agent):
        frame = inspect.currentframe().f_back
        assert frame.f_code.co_name == 'non_agent_step' and frame.f_lineno in (643,724)
        capture = frame.f_lineno == 724
        deaths.append(dict(t=round(e.time+.1,2),id=agent.agent_id,captured=capture,
                           remaining_energy=round(agent.energy,2)))
        return native_kill(agent)
    def fruit_observer(fruit):
        frame = inspect.currentframe().f_back
        assert frame.f_code.co_name == 'non_agent_step' and frame.f_lineno in (672,733)
        if frame.f_lineno == 672:
            agent = frame.f_locals['agent']
            group = 'bait' if agent.agent_id in bait_ids else 'worker'
            old = food_energy_cursor[agent.agent_id]
            # Cursor is initialized before aging. Subtract native passive/old-age
            # charges only for the first fruit eaten by this agent this tick.
            if agent.agent_id not in ate_this_tick:
                old -= .1
                if agent.age > agent.max_age:
                    old -= .01*agent.age
            ate_this_tick.add(agent.agent_id)
            food_energy_cursor[agent.agent_id] = agent.energy
            food_count[group] += 1
            food_energy[group] += fruit.energy
            food_received[group] += max(0.,agent.energy-old)
        return native_remove(fruit)
    e.kill_agent, e.remove_fruit = kill_observer, fruit_observer
    for pid, predator in enumerate(e.predators):
        native_step = predator.step
        def predator_observer(observation, *, pid=pid, native_step=native_step):
            nonlocal worker_target_ticks, first_gap
            seen = [o for o in observation if o['type']=='Agent']
            target = min(seen,key=lambda o:o['distance'])['id'] if seen else None
            selected[pid] = target
            pred_active[pid] = pred_active.get(pid,0)+1
            pred_occupied[pid] = pred_occupied.get(pid,0)+(target in bait_ids)
            worker_target_ticks += target in worker_ids
            if target in bait_ids:
                pred_gap[pid] = 0
                if last_target.get(pid) in bait_ids and last_target[pid] != target:
                    switches.append(dict(t=round(e.time,2),predator=pid,
                                         outgoing=last_target[pid],incoming=target))
            else:
                pred_gap[pid] = pred_gap.get(pid,0)+1
                pred_max_gap[pid] = max(pred_max_gap.get(pid,0),pred_gap[pid])
                if first_gap is None:
                    first_gap = round(e.time,2)
            last_target[pid] = target
            return native_step(observation)
        predator.step = predator_observer

    for _ in range(round(seconds*10)):
        action_time = e.time
        inputs = states
        before = dict(e.agents_dict)
        actions, decisions = ctrl(states,e.time)
        assert len(actions)==len({a['agent_id'] for a in actions})==len(states)
        assert {a['agent_id'] for a in actions} == set(before)
        for action in actions:
            aid = action['agent_id']
            a = before[aid]
            old = a.energy
            assert all(math.isfinite(action[k]) for k in ('move_distance','move_direction','turn_angle'))
            e.agent_step(aid,action['move_distance'],action['move_direction'],
                         action['turn_angle'],action['spawn_agent'])
            new = [a for a in e.agents if a.agent_id not in known_ids]
            birth_cost += 100*len(new)
            for child in new:
                known_ids.add(child.agent_id)
                bait_ids.add(child.agent_id)
                births.append(dict(t=round(e.time,2),id=child.agent_id,parent=aid,
                    energy=child.energy,speed=child.speed,sprint_speed=child.sprint_speed,
                    max_energy=child.max_energy))
            spent['bait' if aid in bait_ids else 'worker'] += old-a.energy-100*len(new)
        selected = [None]*predators
        ate_this_tick = set()
        food_energy_cursor = {a.agent_id:a.energy for a in e.agents}
        death_cursor = len(deaths)
        e.non_agent_step(.1)
        for death in deaths[death_cursor:]:
            if death['captured']:
                captured_ids.add(death['id'])
                captured_energy += death['remaining_energy']
                if first_capture is None:
                    first_capture = death['t']
        population_seconds += len(e.agents)*.1
        worker_seconds += sum(i in e.agents_dict for i in worker_ids)*.1
        peak_population = max(peak_population,len(e.agents))
        for a in e.agents:
            threshold = max(ctrl.newborn_ready,a.max_energy/5+25)
            if (a.agent_id in bait_ids and a.agent_id not in prepared and
                any(b['id']==a.agent_id for b in births) and a.energy>=threshold):
                prepared.add(a.agent_id)
                ready_events.append(dict(t=round(e.time,2),id=a.agent_id,
                                         energy=round(a.energy,2),threshold=round(threshold,2)))
        for a in before.values():
            if a.agent_id in bait_ids:
                min_separation = min(min_separation,*(math.hypot(a.x-p.x,a.y-p.y) for p in e.predators))
        states = [e.get_agent_state(a.agent_id) for a in e.agents]
        if recorder:
            recorder.capture([(a['agent_id'],a) for a in actions],decisions,inputs,action_time)
        if round(e.time*10)%10==1:
            trace.append(dict(t=round(e.time,1),
                agents=[[a.agent_id,round(a.x,1),round(a.y,1),round(a.energy,1)] for a in e.agents],
                predators=[[round(p.x,1),round(p.y,1),round(p.energy,1),p.resting] for p in e.predators],
                target=selected,active=ctrl.active,incoming=ctrl.incoming))
    active = sum(pred_active.values())
    occupied = sum(pred_occupied.values())
    result = dict(parameters=parameters, **hashes, measurement='native delegated event observers',
        active_ticks=active,occupied_ticks=occupied,active_retention=round(occupied/max(active,1),5),
        per_predator_retention={i:round(pred_occupied[i]/n,5) for i,n in pred_active.items()},
        worker_target_ticks=worker_target_ticks,
        longest_active_gap_seconds=round(max(pred_max_gap.values(),default=0)*.1,2),
        per_predator_longest_gap={k:round(v*.1,2) for k,v in pred_max_gap.items()},
        actual_target_switches=len(switches),confirmed_target_changes=switches,
        baits_alive=sum(i in e.agents_dict for i in bait_ids),
        workers_alive=sum(i in e.agents_dict for i in worker_ids),
        worker_seconds=round(worker_seconds,2),population_seconds=round(population_seconds,2),
        peak_population=peak_population,bait_captures=len(captured_ids&bait_ids),
        worker_captures=len(captured_ids&set(worker_ids)),deaths=deaths,
        first_active_target_gap=first_gap,first_capture=first_capture,
        min_bait_predator_separation=round(min_separation,2) if baits else None,
        birth_cost=birth_cost,births=births,newborn_prepared=ready_events,
        captured_agent_energy=round(captured_energy,2),food_count=food_count,
        food_energy={k:round(v,2) for k,v in food_energy.items()},
        food_received={k:round(v,2) for k,v in food_received.items()},
        food_waste={k:round(food_energy[k]-food_received[k],2) for k in food_energy},
        movement_turn_energy={k:round(v,2) for k,v in spent.items()},
        energy_remaining={a.agent_id:round(a.energy,2) for a in e.agents},events=ctrl.events,trace=trace)
    if recorder:
        recorder.save(OUT/(record+'.json.gz'),overwrite=True)
        result['replay'] = record+'.json.gz'
    return result


def save(name,runs):
    payload = dict(source_commit=SOURCE,controller_sha256=digest(Path(__file__).parent/'controller.py'),
                   runner_sha256=digest(__file__),runs=runs)
    (OUT/(name+'.json')).write_text(json.dumps(payload,indent=2,allow_nan=False)+'\n')


if __name__=='__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--phase',choices=['pilot'],default='pilot')
    args = parser.parse_args()
    runs = []
    for policy in ('radial','single','mpc'):
        for energy in (150,500):
            row = run(policy=policy,energy=energy,seconds=30)
            runs.append(row)
            print(json.dumps({k:v for k,v in row.items() if k not in ('trace','events')}),flush=True)
    save(args.phase,runs)
