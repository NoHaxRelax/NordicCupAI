"""Local controlled relay experiments; source engine is never modified."""
from __future__ import annotations
import argparse,hashlib,json,math,os,sys,inspect
from pathlib import Path
ROOT=Path(__file__).resolve().parents[2]
sys.path[:0]=[str(ROOT/'research'),str(ROOT/'debugger'),str(Path(__file__).parent)]
from sprinter_decoys import fixture,add_agent,add_pred,observe,ripe_food
from controller import RelayController,MultiRelayController
from recorder import ReplayRecorder
from src.elements.tree import Tree
from src.elements.environment import Environment
import types

OUT=ROOT/'results/sprinter_relay'

def run(*,seed=0,energy=150,gap=60,heading=0.,pred_energy=102.,seconds=45,policy='mpc',reserve_angle=math.pi,reserve_gap=60,
        food='none',workers=0,worker_near=False,predators=1,pairs=1,record=None,**params):
    e=fixture(seed);p=add_pred(e,x=800,y=600,heading=heading,energy=pred_energy)
    a=add_agent(e,800+gap,600,energy);baits=[a]
    if policy not in ('single','orbit_single','none'):
        b=add_agent(e,800+reserve_gap*math.cos(reserve_angle),600+reserve_gap*math.sin(reserve_angle),energy)
        b.direction=reserve_angle+math.pi;baits.append(b)
    if policy=='none':e.kill_agent(a);baits=[]
    groups=[[a.agent_id for a in baits]]
    if predators>1:
        add_pred(e,x=800,y=950,heading=heading,energy=pred_energy)
        if pairs>1 and policy!='none':
            aa=add_agent(e,800+gap,950,energy);baits.append(aa);group=[aa.agent_id]
            if policy not in ('single','orbit_single'):
                bb=add_agent(e,800+reserve_gap*math.cos(reserve_angle),950+reserve_gap*math.sin(reserve_angle),energy);bb.direction=reserve_angle+math.pi;baits.append(bb);group.append(bb.agent_id)
            groups.append(group)
    worker_ids=[]
    for i in range(workers):
        a=add_agent(e,1050 if worker_near else 1200,500+70*i,150);a.direction=math.pi;worker_ids.append(a.agent_id)
    if food=='finite':
        ripe_food(e,[(x,y) for x,y in [(740,600),(860,600),(800,660),(800,540),(680,600),(920,600),(800,720),(800,480)]]+[(1200,500+70*i) for i in range(workers)])
    if food in ('trees','renewing'):
        for x,y in [(740,600),(860,600),(800,660),(800,540),(680,600),(920,600)]+[(1200,500+70*i) for i in range(workers)]:
            tree=Tree(x,y);tree.grow(20);e.trees.append(tree)
        if predators>1 and pairs>1:
            for x,y in [(740,950),(860,950),(800,1010),(800,890),(680,950),(920,950)]:
                tree=Tree(x,y);tree.grow(20);e.trees.append(tree)
        e._update_tree_grid()
        if food=='renewing':e.spawn_tree=types.MethodType(Environment.spawn_tree,e)
    ctrl=RelayController([a.agent_id for a in baits],variant='orbit' if policy=='orbit_single' else 'mpc' if policy in ('single','none') else policy,**params)
    if len(groups)>1:ctrl=MultiRelayController(groups,variant='orbit' if policy in ('orbit','orbit_single') else policy,**params)
    states=observe(e);rec=None
    if record:
        rec=ReplayRecorder(e,title=record,policy=policy,seed=seed,every=2,scenario='arranged relay',notes='Observation-only controller; arranged clear forest fixture. Native energy, food, aging, predator sensing and physics.');rec.capture(force=True)
    ids=set(a.agent_id for a in baits);births=[];initial_ids=set(e.agents_dict);deaths=[];trace=[];targets=[];last={};handoffs=0
    active=occupied=worker_target=0;gap_run=max_gap=0;capture_ids=set();food_count={'bait':0,'worker':0};food_energy={'bait':0.,'worker':0.};spent={'bait':0.,'worker':0.};birth_cost=0.;worker_seconds=0;population_seconds=0.;peak_population=len(e.agents);food_waste={'bait':0.,'worker':0.};captured_energy=0.;min_separation=10000.;first_gap=None;first_capture=None;confirmed_changes=[];pred_gap={};pred_max_gap={};newborn_ready=[];prepared=set();diagnostic_baits=set(ids);pred_active={};pred_occupied={};diagnostic_frames={};max_odometry_error=0.
    for tick in range(round(seconds*10)):
        inputs=states;before=dict(e.agents_dict);actions,decisions=ctrl(states,e.time)
        for act in actions:
            aid=act['agent_id'];a=before[aid];old=a.energy
            e.agent_step(aid,act['move_distance'],act['move_direction'],act['turn_angle'],act['spawn_agent'])
            for child in e.agents:
                if child.agent_id not in initial_ids:
                    initial_ids.add(child.agent_id);diagnostic_baits.add(child.agent_id)
                    births.append(dict(t=round(e.time,2),id=child.agent_id,energy=child.energy,parent=aid,speed=child.speed,sprint_speed=child.sprint_speed,max_energy=child.max_energy))
            reproduction_cost=100 if act['spawn_agent'] and a.energy<old-90 else 0
            birth_cost+=reproduction_cost
            spent['bait' if aid in ctrl.baits else 'worker']+=old-a.energy-reproduction_cost
        selected=[]
        for pid,p in enumerate(e.predators):
            obs=p.observe(agents=e._get_local_agents(p),edges=e._get_local_edges(p));seen=[o for o in obs if o['type']=='Agent'];target=min(seen,key=lambda o:o['distance'])['id'] if seen else None
            # Wake tick executes if resting energy already exceeds100.
            awake=not p.resting or p.energy>p.max_energy*.5
            if awake:
                pred_active[pid]=pred_active.get(pid,0)+1;pred_occupied[pid]=pred_occupied.get(pid,0)+(target in diagnostic_baits)
                active+=1;occupied+=target in diagnostic_baits;worker_target+=target in worker_ids
                if target in diagnostic_baits:
                    pred_gap[pid]=0
                    if last.get(pid) in diagnostic_baits and last[pid]!=target:
                        handoffs+=1;confirmed_changes.append(dict(t=round(e.time,2),predator=pid,outgoing=last[pid],incoming=target))
                    gap_run=0
                else:
                    gap_run+=1;max_gap=max(max_gap,gap_run);pred_gap[pid]=pred_gap.get(pid,0)+1;pred_max_gap[pid]=max(pred_max_gap.get(pid,0),pred_gap[pid])
                    if first_gap is None:first_gap=round(e.time,2)
                last[pid]=target
            selected.append(target)
        # Predict food attribution in native agent order (there are no moves during this pass).
        available=list(e.fruits)
        for a in e.agents:
            if a.energy<=.1:continue
            for f in list(available):
                if math.hypot(a.x-f.x,a.y-f.y)<a.size+f.radius:
                    group='bait' if a.agent_id in diagnostic_baits else 'worker';food_count[group]+=1;food_energy[group]+=f.energy;food_waste[group]+=max(0,a.energy+f.energy-a.max_energy);available.remove(f)
        energy_before={aid:a.energy for aid,a in before.items()}
        e.non_agent_step(.1)
        for a in e.agents:
            if a.agent_id not in initial_ids:raise AssertionError('Untracked newborn')
        for aid,a in before.items():
            if aid not in e.agents_dict:
                captured=a.energy>0
                if captured:
                    capture_ids.add(aid);captured_energy+=a.energy
                    if first_capture is None:first_capture=round(e.time,2)
                deaths.append(dict(t=round(e.time,2),id=aid,captured=captured,remaining_energy=round(a.energy,2)))
        population_seconds+=len(e.agents)*.1;peak_population=max(peak_population,len(e.agents))
        for a in e.agents:
            if a.agent_id in diagnostic_baits and any(b['id']==a.agent_id for b in births) and a.agent_id not in prepared and a.energy>=ctrl.newborn_ready:
                prepared.add(a.agent_id);newborn_ready.append(dict(t=round(e.time,2),id=a.agent_id,energy=round(a.energy,2)))
        for a in before.values():
            if a.agent_id in diagnostic_baits:
                min_separation=min(min_separation,*(math.hypot(a.x-p.x,a.y-p.y) for p in e.predators))
        worker_seconds+=sum(i in e.agents_dict for i in worker_ids)*.1
        states=[e.get_agent_state(a.agent_id) for a in e.agents]
        if rec:rec.capture([(a['agent_id'],a) for a in actions],decisions,inputs,e.time-.1)
        if tick%10==0:trace.append(dict(t=round(e.time,1),agents=[[a.agent_id,round(a.x,1),round(a.y,1),round(a.energy,1)] for a in e.agents],predators=[[round(p.x,1),round(p.y,1),round(p.energy,1),p.resting] for p in e.predators],target=selected,active=ctrl.active,incoming=ctrl.incoming))
    result=dict(parameters=dict(seed=seed,energy=energy,gap=gap,heading=heading,pred_energy=pred_energy,seconds=seconds,policy=policy,reserve_angle=reserve_angle,reserve_gap=reserve_gap,food=food,workers=workers,worker_near=worker_near,predators=predators,pairs=pairs,**params),
        active_ticks=active,occupied_ticks=occupied,active_retention=round(occupied/max(active,1),5),worker_target_ticks=worker_target,longest_active_gap_seconds=round(max(pred_max_gap.values(),default=0)*.1,2),actual_target_switches=handoffs,
        baits_alive=sum(i in e.agents_dict for i in diagnostic_baits),workers_alive=sum(i in e.agents_dict for i in worker_ids),worker_seconds=round(worker_seconds,2),bait_captures=len(capture_ids&diagnostic_baits),worker_captures=len(capture_ids&set(worker_ids)),deaths=deaths,
        per_predator_retention={pid:round(pred_occupied[pid]/n,5) for pid,n in pred_active.items()},first_active_target_gap=first_gap,first_capture=first_capture,min_bait_predator_separation=round(min_separation,2) if baits else None,per_predator_longest_gap={k:round(v*.1,2) for k,v in pred_max_gap.items()},confirmed_target_changes=confirmed_changes,birth_cost=birth_cost,peak_population=peak_population,population_seconds=round(population_seconds,2),captured_agent_energy=round(captured_energy,2),food_waste={k:round(v,2) for k,v in food_waste.items()},newborn_prepared=newborn_ready,births=births,food_count=food_count,food_energy={k:round(v,2) for k,v in food_energy.items()},movement_turn_energy={k:round(v,2) for k,v in spent.items()},energy_remaining={a.agent_id:round(a.energy,2) for a in e.agents},events=ctrl.events,trace=trace)
    if rec:rec.save(OUT/(record+'.json.gz'),overwrite=True);result['replay']=record+'.json.gz'
    return result

def save(name,runs):
    payload=dict(source_commit='acfc31a4003a5f91bf11032a02cd98c178ddbd7e',controller_sha256=hashlib.sha256(Path(inspect.getfile(RelayController)).read_bytes()).hexdigest(),runner_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),runs=runs)
    (OUT/(name+'.json')).write_text(json.dumps(payload,indent=2)+'\n')

if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('--phase',default='pilot');args=parser.parse_args()
    runs=[]
    if args.phase=='pilot':
        cases=[dict(policy=p,energy=e,seconds=30) for p in ('radial','single','mpc') for e in (150,500)]
    else:raise ValueError(args.phase)
    for case in cases:
        row=run(**case);runs.append(row);print(json.dumps({k:v for k,v in row.items() if k not in ('trace','events')}),flush=True)
    save(args.phase,runs)
